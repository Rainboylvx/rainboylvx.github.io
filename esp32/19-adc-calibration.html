本课完成后能做什么

  • 旋转 RV1，raw 和 mV 随位置变化。
  • 选择低、中、高三个未饱和位置，与万用表比较并记录误差。
  • 拆掉跳线后认识悬空抖动，不能解释为有效传感器数据。

旋钮位置如何变成一个整数

RV1 两端连接 3.3V 和 GND，滑动端经 P3 跳线送到 GPIO8。ADC 对输入电压采样，给出原始整数。GPIO8 的 ADC 功能是 ADC1_CHANNEL_7，GPIO 编号与通道号不能混用。

P3 同时承担红外发射切换。采集电压时接 RV↔AIN，红外发射时接 AIN↔RMT；两课不能同时让 GPIO8 当模拟输入和数字输出。

为什么不直接 raw × 3300 ÷ 4095

位宽说明码值数量，衰减决定输入量程，参考电压和芯片误差影响转换关系。原始满量程并不等于精确的 3.300V。ESP32-S3 在本例使用 12 dB 衰减，其有效测量范围也不等于电源轨，旋钮接近顶端可能饱和。

工程使用 oneshot 驱动，建立与通道、衰减和位宽匹配的曲线拟合校准句柄，再调用 adc_cali_raw_to_voltage() 获得毫伏。若校准不可用，只报告 raw，不把一个估算值伪装成已校准值。驱动参考见文末官方文档。

平均值能消除哪些问题

每轮读取 16 次，间隔 1 ms，再求整数平均。它可减轻部分随机噪声，但无法修复引脚接错、量程饱和或系统性偏差。采样失败的这一轮不输出新结果，避免拿未赋值变量继续计算。

实板验证应把旋钮固定在几个位置，用万用表同时量 GPIO8 对地电压，与日志的 mV 比较；不要用“日志数字在变化”替代准确度验证。

工程如何组织

components/BSP/ADC/voltage.c 拥有 ADC 和校准句柄，负责采样平均与单位转换；main/main.c 只负责周期和显示。calibrated 明确告诉应用毫伏值是否可信。

固定引脚以 CONFIG_LESSON_* 常量写入本工程 Kconfig.projbuild，构建后可在 sdkconfig 核对；这些默认值对应本板连线，不表示任意改数字就能改变硬件。

应用代码：入口与处理流程

下面代码直接取自配套工程的 main/main.c，头文件和 BSP 实现在仓库中一并提供。初学时先看 app_main()，再追踪它调用的模块。

 1#include "voltage.h"
 2#include "esp_log.h"
 3#include "freertos/FreeRTOS.h"
 4#include "freertos/task.h"
 5void app_main(void)
 6{
 7    ESP_ERROR_CHECK(voltage_init());
 8    while(1) {
 9        int raw,mv; bool calibrated;
10        esp_err_t err=voltage_read(&raw,&mv,&calibrated);
11        if(err!=ESP_OK) ESP_LOGW("adc","%s",esp_err_to_name(err));
12        else if(calibrated) ESP_LOGI("adc","raw=%d voltage=%d mV",raw,mv);
13        else ESP_LOGI("adc","raw=%d (uncalibrated)",raw);
14        vTaskDelay(pdMS_TO_TICKS(200));
15    }
16}

BSP 采样与校准实现

 1#include "voltage.h"
 2#include "sdkconfig.h"
 3#include "esp_adc/adc_oneshot.h"
 4#include "esp_adc/adc_cali.h"
 5#include "esp_adc/adc_cali_scheme.h"
 6#include "esp_check.h"
 7#include "freertos/FreeRTOS.h"
 8#include "freertos/task.h"
 9static adc_oneshot_unit_handle_t s_unit;
10static adc_cali_handle_t s_calibration;
11esp_err_t voltage_init(void)
12{
13    adc_oneshot_unit_init_cfg_t u={.unit_id=ADC_UNIT_1};
14    ESP_RETURN_ON_ERROR(adc_oneshot_new_unit(&u,&s_unit),"adc","unit");
15    adc_oneshot_chan_cfg_t channel={.atten=ADC_ATTEN_DB_12,.bitwidth=ADC_BITWIDTH_DEFAULT};
16    ESP_RETURN_ON_ERROR(adc_oneshot_config_channel(s_unit,CONFIG_LESSON_ADC_CHANNEL,&channel),"adc","channel");
17    adc_cali_curve_fitting_config_t c={.unit_id=ADC_UNIT_1,.chan=CONFIG_LESSON_ADC_CHANNEL,
18        .atten=ADC_ATTEN_DB_12,.bitwidth=ADC_BITWIDTH_DEFAULT};
19    esp_err_t result=adc_cali_create_scheme_curve_fitting(&c,&s_calibration);
20    if(result!=ESP_OK) ESP_LOGW("adc","Calibration unavailable: %s; raw only",esp_err_to_name(result));
21    return ESP_OK; // 校准不可用时仍允许读取 raw。
22}
23esp_err_t voltage_read(int *raw,int *millivolts,bool *calibrated)
24{
25    int sum=0;
26    for(int i=0;i<16;i++) {
27        int value;
28        ESP_RETURN_ON_ERROR(adc_oneshot_read(s_unit,CONFIG_LESSON_ADC_CHANNEL,&value),"adc","read");
29        sum+=value;
30        vTaskDelay(pdMS_TO_TICKS(1)); // 分散采样并让出 CPU。
31    }
32    int average=sum/16,mv=0;
33    bool valid=s_calibration && adc_cali_raw_to_voltage(s_calibration,average,&mv)==ESP_OK;
34    *raw=average; *millivolts=mv; *calibrated=valid;
35    return ESP_OK;
36}

2idf_component_register(SRCS "main.c" INCLUDE_DIRS "." PRIV_REQUIRES BSP)
