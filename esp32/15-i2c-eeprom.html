EEPROM 与普通变量有什么不同

RAM 中的变量断电后丢失，AT24C02 的 EEPROM 在断电后保留数据。它适合保存变化不频繁的小配置，例如上次使用的模式，不适合把每次循环的传感器值都写进去。写入需要时间，也受擦写寿命限制。

AT24C02 有 256 个字节地址，范围 0～255。本例使用 7 位 I²C 地址 0x50，不要把旧资料中的写地址 0xA0 直接传给新驱动。板卡的 A0～A2 和 WP 接地，分别确定地址和允许写入。

跨页写入为什么必须拆包

EEPROM 的页写缓冲只有 8 字节。一次事务越过页尾时，不会自动顺序写到下一页，还可能在同一页内回绕，覆盖已有数据。

例如，从地址 6 保存含字符串结束符的 15 字节数据，需要拆成：

起点本次字节数写入范围
626～7
888～15
16516～20

计算式为 min(剩余字节数, 8 - 地址 % 8)。每一包先发送存储地址，再发送数据。总线事务结束不等于内部写周期结束，因此每包之后用有上限的 ACK 轮询等待器件重新应答。

怎样证明真的保存了，而不是刚写又读

程序启动时先读地址 6 并输出十六进制字节，不自动写入。按 BOOT 才保存 EEPROM-PERSIST\0，随后读回用 memcmp() 校验。校验使用固定长度，不依赖未初始化存储里是否恰好存在字符串结束符。

第一次上电读出 0xff 或旧内容都正常。完成写入后断电重启，不按 BOOT，确认启动日志仍有同样字节，才能证明持久化。本例保存的是固定内容，未实现事务数据库。若产品要求断电写入也能恢复，应增加版本、长度、CRC 和双槽提交策略。

工程如何组织

驱动放在 components/BSP/，应用放在 main/main.c。本工程包含 BUS、EEPROM 模块。ESP-IDF 会自动发现项目根目录下的 components/BSP。main 通过 PRIV_REQUIRES BSP 声明组件依赖，INCLUDE_DIRS 则导出编译时使用的头文件搜索路径，不涉及运行时文件查找。

固定引脚通过 CONFIG_LESSON_* 常量定义在本工程的 Kconfig.projbuild 中，构建后可在 sdkconfig 中核对。这些默认值对应本板连线，仅修改配置数字并不能改变硬件连接。

应用代码：入口与处理流程

下面代码直接取自配套工程的 main/main.c，头文件和 BSP 实现在仓库中一并提供。初学时先看 app_main()，再追踪它调用的模块。

 1#include "sdkconfig.h"
 2#include <stdio.h>
 3#include <stdint.h>
 4#include <stdbool.h>
 5#include <string.h>
 6#include <stdlib.h>
 7#include <assert.h>
 8#include "esp_err.h"
 9#include "esp_log.h"
10#include "esp_timer.h"
11#include "freertos/FreeRTOS.h"
12#include "freertos/task.h"
13#include "eeprom.h"
14#include "driver/gpio.h"
15void app_main(void)
16{
17    i2c_master_bus_handle_t bus;
18    ESP_ERROR_CHECK(bus_init(0,CONFIG_LESSON_I2C_SDA,CONFIG_LESSON_I2C_SCL,&bus));
19    ESP_ERROR_CHECK(eeprom_open(bus));
20    gpio_config_t key={.pin_bit_mask=1ULL<<CONFIG_LESSON_KEY,.mode=GPIO_MODE_INPUT,.pull_up_en=true};
21    ESP_ERROR_CHECK(gpio_config(&key));
22    const char message[]="EEPROM-PERSIST"; // 含结尾 0，共 15 字节，从地址 6 开始跨三页。
23    char data[sizeof(message)];
24    esp_err_t err=eeprom_read(6,data,sizeof(data));
25    if(err==ESP_OK) ESP_LOG_BUFFER_HEX("saved",data,sizeof(data));
26    ESP_LOGI("eeprom","Press BOOT to write; power cycle to inspect saved bytes");
27    bool previous=false;
28    while(1) {
29        bool pressed=gpio_get_level(CONFIG_LESSON_KEY)==0;
30        if(pressed && !previous) {
31            vTaskDelay(pdMS_TO_TICKS(30));
32            if(gpio_get_level(CONFIG_LESSON_KEY)==0) {
33                err=eeprom_write(6,message,sizeof(message));
34                if(err==ESP_OK) err=eeprom_read(6,data,sizeof(data));
35                if(err==ESP_OK && memcmp(data,message,sizeof(data))) err=ESP_ERR_INVALID_RESPONSE;
36                ESP_LOGI("eeprom","write + verify: %s",esp_err_to_name(err));
37            }
38        }
39        previous=pressed;
40        vTaskDelay(pdMS_TO_TICKS(20));
41    }
42}

BSP 关键实现：EEPROM

以下是本课对应驱动的完整源文件。对外声明见 eeprom.h。公共 BUS 模块统一使用新 I²C master API，将单次事务等待限制在 50 ms；设备 API 使用 7 位地址。

 1#include "eeprom.h"
 2#include <string.h>
 3#include "freertos/FreeRTOS.h"
 4#include "freertos/task.h"
 5static i2c_master_dev_handle_t s_dev;
 6static i2c_master_bus_handle_t s_bus;
 7esp_err_t eeprom_open(i2c_master_bus_handle_t bus)
 8{
 9    s_bus=bus;
10    return bus_device(bus,0x50,&s_dev);
11}
12esp_err_t eeprom_read(size_t address, void *data, size_t n)
13{
14    if(!data || !n || address>=256 || n>256-address) return ESP_ERR_INVALID_ARG;
15    return reg_read(s_dev,(uint8_t)address,data,n);
16}
17esp_err_t eeprom_write(size_t address, const void *data, size_t n)
18{
19    if(!data || !n || address>=256 || n>256-address) return ESP_ERR_INVALID_ARG;
20    const uint8_t *p=data;
21    while(n) {
22        size_t chunk=8-address%8; // AT24C02 一页只有 8 字节，不能跨页发一次写命令。
23        if(chunk>n) chunk=n;
24        uint8_t packet[9]={(uint8_t)address};
25        memcpy(packet+1,p,chunk);
26        ESP_RETURN_ON_ERROR(i2c_master_transmit(s_dev,packet,chunk+1,50),"eeprom","write");
27        esp_err_t ready=ESP_FAIL;
28        // 写周期内芯片 NACK 是正常忙状态；最多等约 20 ms，不能无限等。
29        for(int attempt=0;attempt<10;attempt++) {
30            vTaskDelay(pdMS_TO_TICKS(2));
31            ready=i2c_master_probe(s_bus,0x50,2);
32            if(ready==ESP_OK) break;
33        }
34        if(ready!=ESP_OK) return ready;
35        address+=chunk; p+=chunk; n-=chunk;
36    }
37    return ESP_OK;
38}
