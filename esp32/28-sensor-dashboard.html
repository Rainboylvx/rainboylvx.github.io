本课将前面的总线、传感器、FreeRTOS 队列和显示知识组合起来。配套完整工程：examples/28-sensor-dashboard。

环境与验收边界： DNESP32S3 V1.2，ESP-IDF v5.5.5；本机 macOS 构建通过。Ubuntu 26、外设接线、屏幕效果和实际测量仍待实板验收。下文输出均为预期格式，不能当作已采集的硬件日志。

本课完成后能做什么

  • 三个来源按不同周期更新，屏幕显示字段与串口一致。
  • 断开 OLED 后串口采样继续；恢复后显示重试成功。
  • 将一个传感器地址临时改错，仅该来源显示 ERR，其他来源保持更新。
  • 把共享 SDA 故障与单设备地址错误分开测试，说明影响范围。
  • 长时间观察队列满告警和数据年龄，不仅验证第一帧。

综合项目先分清谁负责什么

任务周期负责的状态
光照采样200 msAP3216C 初始化与读数
加速度采样100 msQMA6100P 初始化与读数
芯片温度采样1000 ms温度读取
应用/显示约 200 ms 刷新最近读数、错误和时间戳

采集任务只发送事件，不操作 OLED。应用任务是显示模型的唯一写入者。屏幕显示 ALS 原始值、X 轴 g 和芯片温度，串口每两秒输出状态和数据年龄。它是阶段演示界面，不宣称环境温度或完整运动姿态。

事件必须携带来源、结果和时间

sample_event_t 包含传感器编号、错误码、采样时间和三个浮点字段。FreeRTOS 队列按值复制结构体，因此任务下一轮复用局部变量不会破坏已入队的数据。不要把临时局部变量的指针发给消费者。

成功、失败、还没收到数据、数据过期是四种不同状态。显示中对应数值、ERR、WAIT、STALE；最近一次失败不会被前一次成功值遮住。3 秒未更新才标记过期，单调时间不受系统校时影响。

如何让一个设备异常不阻止其他设备

总线和设备句柄在启动阶段建立，实际器件初始化移到对应任务。读失败后降为每秒重试；每笔 I²C 事务有 50 ms 上限。显示失败也只进入屏幕自己的限频重试，不终止采样任务。

但“软件隔离”不能突破电气事实：AP3216C 和 QMA6100P 共用同一物理 I²C 总线，某个器件把 SDA 永久拉低时，两者都可能失败。OLED 用另一条总线，芯片温度不依赖 I²C，仍可继续。验证时分别测试缺少单个地址和整条线被拉低，理解两种故障范围。

队列容量不是越大越好

正常每秒约 16 个事件（5 次光照、10 次加速度、1 次温度），32 项队列为短暂显示操作提供余量。采集发送最多等待 10 ms，队列满会记录警告并丢弃这一条新样本，不阻塞整个采样链。

应用一次消费一个事件，最多等 20 ms，再检查刷新期限。绘屏仍是同步传输，100 kHz 下需要约百毫秒，这段时间采集可以继续入队。程序不保证严格实时刷新；若后续加入更多设备，应测量队列压力并采用快照、合并更新或更快总线。

vTaskDelayUntil() 按上次唤醒点规划下一次执行，减少周期累积漂移；它不会让一次超过周期的 I²C 操作自动变快。

工程如何组织

驱动放在 components/BSP/，应用放在 main/main.c。本工程包含 BUS、EXIO、OLED、AP3216C、QMA6100P、TEMP 模块。项目根目录的 components/BSP 会被 ESP-IDF 自动发现，main 的 PRIV_REQUIRES BSP 声明组件依赖；INCLUDE_DIRS 导出头文件搜索路径，不是运行时查找文件。

固定引脚以 CONFIG_LESSON_* 常量写入本工程 Kconfig.projbuild，构建后可在 sdkconfig 核对；这些默认值对应本板连线，不表示任意改数字就能改变硬件。

应用代码：入口与处理流程

下面代码直接取自配套工程的 main/main.c，头文件和 BSP 实现在仓库中一并提供。初学时先看 app_main()，再追踪它调用的模块。

  1#include "sdkconfig.h"
  2#include <stdio.h>
  3#include <stdint.h>
  4#include <stdbool.h>
  5#include <string.h>
  6#include <stdlib.h>
  7#include <assert.h>
  8#include "esp_err.h"
  9#include "esp_log.h"
 10#include "esp_timer.h"
 11#include "freertos/FreeRTOS.h"
 12#include "freertos/task.h"
 13#include "ap3216c.h"
 14#include "qma6100p.h"
 15#include "temp.h"
 16#include "oled.h"
 17#include "exio.h"
 18#include "freertos/queue.h"
 19typedef enum { LIGHT, MOTION, CHIP, SENSOR_COUNT } sensor_id_t;
 20typedef struct {
 21    sensor_id_t id;
 22    esp_err_t error;
 23    int64_t sampled_us;
 24    float x,y,z;
 25} sample_event_t;
 26static QueueHandle_t s_samples;
 27static void sample_task(void *arg)
 28{
 29    sensor_id_t id=(sensor_id_t)(intptr_t)arg;
 30    const int periods[]={200,100,1000};
 31    bool ready=id==CHIP;
 32    TickType_t wake=xTaskGetTickCount();
 33    while(1) {
 34        sample_event_t e={.id=id,.sampled_us=esp_timer_get_time()};
 35        if(!ready) {
 36            e.error=id==LIGHT ? light_reset() : motion_reset();
 37            ready=e.error==ESP_OK;
 38        }
 39        if(ready) {
 40            if(id==LIGHT) {
 41                light_sample_t v;
 42                e.error=light_read(&v);
 43                if(e.error==ESP_OK) {
 44                    e.x=v.als; e.y=v.ps; e.z=v.ps_valid;
 45                    if(!v.ps_valid) e.error=ESP_ERR_INVALID_RESPONSE;
 46                }
 47            } else if(id==MOTION) {
 48                acceleration_t v;
 49                e.error=motion_read(&v);
 50                if(e.error==ESP_OK) {e.x=v.x; e.y=v.y; e.z=v.z;}
 51            } else e.error=chip_temp_read(&e.x);
 52            if(e.error!=ESP_OK && id!=CHIP) ready=false;
 53        }
 54        // 队列按值复制；局部变量离开本轮后不会留下悬空指针。
 55        if(xQueueSend(s_samples,&e,pdMS_TO_TICKS(10))!=pdTRUE)
 56            ESP_LOGW("sample","queue full, id=%d",id);
 57        int delay=e.error==ESP_OK ? periods[id] : 1000;
 58        vTaskDelayUntil(&wake,pdMS_TO_TICKS(delay));
 59    }
 60}
 61void app_main(void)
 62{
 63    i2c_master_bus_handle_t sensors,display;
 64    ESP_ERROR_CHECK(bus_init(0,CONFIG_LESSON_I2C_SDA,CONFIG_LESSON_I2C_SCL,&sensors));
 65    ESP_ERROR_CHECK(bus_init(1,CONFIG_LESSON_OLED_SDA,CONFIG_LESSON_OLED_SCL,&display));
 66    // 注册句柄不会要求器件在线；实际通信在各自任务中完成，便于断线后恢复。
 67    ESP_ERROR_CHECK(light_open(sensors));
 68    ESP_ERROR_CHECK(motion_open(sensors));
 69    ESP_ERROR_CHECK(exio_init(sensors));
 70    ESP_ERROR_CHECK(oled_open(display));
 71    ESP_ERROR_CHECK(chip_temp_init());
 72    s_samples=xQueueCreate(32,sizeof(sample_event_t)); assert(s_samples);
 73    for(int id=0;id<SENSOR_COUNT;id++)
 74        ESP_ERROR_CHECK(xTaskCreate(sample_task,"sample",4096,(void *)(intptr_t)id,4,NULL)
 75                        ==pdPASS ? ESP_OK : ESP_ERR_NO_MEM);
 76    sample_event_t latest[SENSOR_COUNT]={0};
 77    bool seen[SENSOR_COUNT]={0},screen_ready=false;
 78    int64_t next_refresh=0,next_retry=0,next_log=0;
 79    while(1) {
 80        sample_event_t e;
 81        if(xQueueReceive(s_samples,&e,pdMS_TO_TICKS(20))==pdTRUE) {
 82            latest[e.id]=e; seen[e.id]=true;
 83        }
 84        int64_t now=esp_timer_get_time();
 85        // 即使屏幕失联，仍持续消费事件；重试限频为 1 秒一次。
 86        if(!screen_ready && now>=next_retry) {
 87            screen_ready=oled_reset()==ESP_OK;
 88            next_retry=now+1000000;
 89        }
 90        if(now>=next_log) {
 91            for(int i=0;i<SENSOR_COUNT;i++)
 92                ESP_LOGI("dashboard","id=%d seen=%d status=%s x=%.2f age=%lld ms",i,seen[i],
 93                    esp_err_to_name(latest[i].error),latest[i].x,(now-latest[i].sampled_us)/1000);
 94            next_log=now+2000000;
 95        }
 96        if(!screen_ready || now<next_refresh) continue;
 97        next_refresh=now+200000;
 98        oled_clear(); oled_text(0,0,"SENSOR DASHBOARD");
 99        const char *labels[]={"ALS","X G","CHIP C"};
100        for(int i=0;i<SENSOR_COUNT;i++) {
101            char text[32];
102            if(!seen[i]) snprintf(text,sizeof(text),"%s WAIT",labels[i]);
103            else if(latest[i].error!=ESP_OK) snprintf(text,sizeof(text),"%s ERR",labels[i]);
104            else if(now-latest[i].sampled_us>3000000) snprintf(text,sizeof(text),"%s STALE",labels[i]);
105            else snprintf(text,sizeof(text),"%s %.2f",labels[i],latest[i].x);
106            oled_text(0,2+i*2,text);
107        }
108        if(oled_refresh()!=ESP_OK) { screen_ready=false; next_retry=now+1000000; }
109    }
110}
