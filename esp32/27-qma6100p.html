静止为什么仍然能读到约 1g

加速度计测量的是包含重力影响的比力。平放且传感器 +Z 朝上时，X、Y 应接近 0g，Z 接近 +1g。三个轴都减成 0，会把计算静态姿态需要的重力方向也一起删掉。

本例先检查芯片 ID，再执行教材复位序列，设置 ±8g 和 100 Hz。输出数据是左对齐的有符号 14 位数；先把高低字节组合为 int16_t，再换算到 g，避免负加速度被误当成巨大的正数。

零偏校准有前提

启动提示后三秒内让板卡静止，确认传感器 +Z 向上，收集 100 个样本取平均。X、Y 的均值作为偏置，Z 的均值减去 1g 作为偏置。

这不是任意姿态自动校准，更不是六面标定。如果板倒放、倾斜或运动，校准就会把重力和运动错记成偏差。测试前核对轴向；需要覆盖未知姿态的产品应另做标定流程及稳定性检测。

从三轴到倾斜角

roll 使用 atan2(y,z)，pitch 使用 atan2(-x,sqrt(y*y+z*z))，弧度再转换成角度。atan2 能保留象限信息，比简单相除后 atan 更方便。

本例任一角绝对值超过 20° 就输出 TILTED，否则 LEVEL。这个阈值适合观察，不是精密姿态控制器；边界处可加入滞回，运动时的加速度也会污染仅靠重力推算的姿态。加速度计无法单独给出可靠的航向 yaw。

工程如何组织

驱动放在 components/BSP/，应用放在 main/main.c。本工程包含 BUS、QMA6100P 模块。项目根目录的 components/BSP 会被 ESP-IDF 自动发现，main 的 PRIV_REQUIRES BSP 声明组件依赖；INCLUDE_DIRS 导出头文件搜索路径，不是运行时查找文件。

应用代码：入口与处理流程

下面代码直接取自配套工程的 main/main.c，头文件和 BSP 实现在仓库中一并提供。初学时先看 app_main()，再追踪它调用的模块。

 1#include <stdio.h>
 2#include <stdint.h>
 3#include <stdbool.h>
 4#include <string.h>
 5#include <stdlib.h>
 6#include <assert.h>
 7#include "esp_err.h"
 8#include "esp_log.h"
 9#include "esp_timer.h"
10#include "freertos/FreeRTOS.h"
11#include "freertos/task.h"
12#include "qma6100p.h"
13#include <math.h>
14void app_main(void)
15{
16    i2c_master_bus_handle_t bus;
17    ESP_ERROR_CHECK(bus_init(0,CONFIG_LESSON_I2C_SDA,CONFIG_LESSON_I2C_SCL,&bus));
18    ESP_ERROR_CHECK(motion_open(bus));
19    ESP_ERROR_CHECK(motion_reset());
20    ESP_LOGI("qma","Keep board STILL, +Z UP for calibration in 3 seconds");
21    vTaskDelay(pdMS_TO_TICKS(3000));
22    acceleration_t bias={0},s;
23    for(int i=0;i<100;i++) {
24        ESP_ERROR_CHECK(motion_read(&s));
25        bias.x+=s.x/100; bias.y+=s.y/100; bias.z+=s.z/100;
26        vTaskDelay(pdMS_TO_TICKS(10));
27    }
28    bias.z-=1; // 保留 +Z 方向的 1g 重力，不能把三个轴的均值都减到零。
29    while(1) {
30        esp_err_t err=motion_read(&s);
31        if(err==ESP_OK) {
32            s.x-=bias.x; s.y-=bias.y; s.z-=bias.z;
33            float roll=atan2f(s.y,s.z)*57.29578f;
34            float pitch=atan2f(-s.x,sqrtf(s.y*s.y+s.z*s.z))*57.29578f;
35            ESP_LOGI("qma","g=(%.3f,%.3f,%.3f) roll=%.1f pitch=%.1f %s",
36                s.x,s.y,s.z,roll,pitch,fabsf(roll)>20 || fabsf(pitch)>20 ? "TILTED":"LEVEL");
37        } else ESP_LOGW("qma","%s",esp_err_to_name(err));
38        vTaskDelay(pdMS_TO_TICKS(200));
39    }
40}

BSP 关键实现：QMA6100P

以下是本课对应驱动的完整源文件。对外声明见 qma6100p.h。公共 BUS 模块统一使用新 I²C master API，将单次事务等待限制在 50 ms；设备 API 使用 7 位地址。

 1#include "qma6100p.h"
 2#include "freertos/FreeRTOS.h"
 3#include "freertos/task.h"
 4static i2c_master_dev_handle_t s_dev;
 5esp_err_t motion_open(i2c_master_bus_handle_t bus) { return bus_device(bus, 0x12, &s_dev); }
 6esp_err_t motion_reset(void)
 7{
 8    uint8_t id;
 9    ESP_RETURN_ON_ERROR(reg_read(s_dev, 0, &id, 1), "qma", "id");
10    if (id != 0x90) return ESP_ERR_NOT_SUPPORTED;
11    ESP_RETURN_ON_ERROR(reg_write(s_dev, 0x36, 0xb6), "qma", "reset");
12    vTaskDelay(pdMS_TO_TICKS(5));
13    ESP_RETURN_ON_ERROR(reg_write(s_dev, 0x36, 0), "qma", "reset release");
14    vTaskDelay(pdMS_TO_TICKS(10));
15    // 教材的芯片上电序列。每一步都检查 I2C 错误。
16    const uint8_t sequence[][2] = {{0x11,0x80},{0x11,0x84},{0x4a,0x20},
17        {0x56,0x01},{0x5f,0x80}};
18    for (size_t i=0; i<sizeof(sequence)/sizeof(sequence[0]); ++i)
19        ESP_RETURN_ON_ERROR(reg_write(s_dev, sequence[i][0], sequence[i][1]), "qma", "setup");
20    vTaskDelay(pdMS_TO_TICKS(1));
21    ESP_RETURN_ON_ERROR(reg_write(s_dev, 0x5f, 0), "qma", "commit");
22    vTaskDelay(pdMS_TO_TICKS(10));
23    ESP_RETURN_ON_ERROR(reg_write(s_dev, 0x0f, 4), "qma", "range"); // ±8 g，1024 LSB/g。
24    ESP_RETURN_ON_ERROR(reg_write(s_dev, 0x10, 0), "qma", "ODR"); // 100 Hz。
25    return reg_write(s_dev, 0x11, 0x84);
26}
27esp_err_t motion_read(acceleration_t *out)
28{
29    uint8_t b[6];
30    ESP_RETURN_ON_ERROR(reg_read(s_dev, 1, b, 6), "qma", "read");
31    // 有符号 14 位数据左对齐；先转 int16_t，避免负值被解释成大正数。
32    out->x = ((int16_t)((b[1]<<8)|b[0]) >> 2) / 1024.0f;
33    out->y = ((int16_t)((b[3]<<8)|b[2]) >> 2) / 1024.0f;
34    out->z = ((int16_t)((b[5]<<8)|b[4]) >> 2) / 1024.0f;
35    return ESP_OK;
36}
