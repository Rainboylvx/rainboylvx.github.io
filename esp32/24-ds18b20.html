一根数据线如何实现双向通信

单总线的数据脚用开漏模式。主机输出 0 表示主动拉低，输出 1 表示释放，总线依靠 4.7 kΩ 上拉变高；传感器需要回传 0 时也可以拉低。若主机用推挽持续输出高，就会与传感器争用总线。

本课默认一个 DS18B20，使用 Skip ROM 0xCC。多器件总线需要枚举和 Match ROM，不能直接让所有器件同时回答。使用外部 3.3V 供电，不支持寄生供电所需的强上拉。

对照 U4 插座丝印和传感器具体封装引脚，不能只按“平面朝前”猜线序。V1.2 的 U4 第 1 脚为 3.3V、第 2 脚为 DQ，第 3、4 脚均连地（第 3 脚对 DHT11 标为 NC）。标准 TO-92 DS18B20 的第 1 脚 GND 接 U4 第 3 脚、第 2 脚 DQ 接 U4 第 2 脚、第 3 脚 VDD 接 U4 第 1 脚；带线封装仍应按其厂商引脚表确认，不能凭线色猜测。

P5 连接 IO0 与 1WIRE_DQ 后，不要再按 BOOT，启动时也不要让总线被拉低。

从复位到温度要经过哪些步骤

主机拉低约 480 μs，再释放，约 70 μs 后采样 Presence。确认器件存在后发送 0xCC、0x44 开始转换；12 位转换最长等待 750 ms，这段长等待让出 CPU，不应放在临界区。

转换完成后再次复位，发送 0xCC、0xBE，读出 9 字节 scratchpad。前两个字节为有符号温度，默认 12 位分辨率下除以 16 得到摄氏度。例如原始 0x0191 对应 25.0625°C；负温度要先转 int16_t，不能当成无符号大整数。

CRC 与超时是读数的一部分

最后一个字节是前 8 字节的 CRC-8。代码还检查保留字节和合理温度范围，排除断线或被拉死后出现的全零假数据。失败时不更新调用者温度。

GPIO 位操作仅用于理解时序，短暂临界区保护每个时隙；750 ms 转换期绝不关中断。Wi-Fi、缓存和长临界区可能影响软件时序，复杂产品可改用 RMT 单总线驱动，本课的实板验证仍不可省略。

工程如何组织

驱动放在 components/BSP/，应用放在 main/main.c。本工程包含 ONEWIRE 模块。项目根目录的 components/BSP 会被 ESP-IDF 自动发现，main 的 PRIV_REQUIRES BSP 声明组件依赖；INCLUDE_DIRS 导出头文件搜索路径，不是运行时查找文件。

固定引脚以 CONFIG_LESSON_* 常量写入本工程 Kconfig.projbuild，构建后可在 sdkconfig 核对；这些默认值对应本板连线，不表示任意改数字就能改变硬件。

BSP 关键实现：ONEWIRE

以下是本课对应驱动的完整源文件。对外声明见 onewire.h。

 1#include "sdkconfig.h"
 2#include "onewire.h"
 3#include "driver/gpio.h"
 4#include "esp_rom_sys.h"
 5#include "freertos/FreeRTOS.h"
 6#include "freertos/task.h"
 7#define DQ CONFIG_LESSON_ONEWIRE
 8static portMUX_TYPE s_lock=portMUX_INITIALIZER_UNLOCKED;
 9esp_err_t ds_init(void)
10{
11    gpio_config_t c={.pin_bit_mask=1ULL<<DQ,.mode=GPIO_MODE_INPUT_OUTPUT_OD,.pull_up_en=true};
12    esp_err_t err=gpio_config(&c);
13    gpio_set_level(DQ,1); // 开漏输出 1 表示释放，靠外部 4.7k 拉高。
14    return err;
15}
16static bool reset(void)
17{
18    gpio_set_level(DQ,0); esp_rom_delay_us(480);
19    portENTER_CRITICAL(&s_lock);
20    gpio_set_level(DQ,1); esp_rom_delay_us(70);
21    bool present=!gpio_get_level(DQ);
22    portEXIT_CRITICAL(&s_lock);
23    esp_rom_delay_us(410);
24    return present && gpio_get_level(DQ); // 排除总线一直短路到地。
25}
26static void write_byte(uint8_t value)
27{
28    for(int bit=0;bit<8;bit++) {
29        portENTER_CRITICAL(&s_lock);
30        gpio_set_level(DQ,0);
31        esp_rom_delay_us(value&1 ? 6 : 60);
32        gpio_set_level(DQ,1);
33        esp_rom_delay_us(value&1 ? 64 : 10);
34        portEXIT_CRITICAL(&s_lock);
35        value>>=1;
36    }
37}
38static uint8_t read_byte(void)
39{
40    uint8_t value=0;
41    for(int bit=0;bit<8;bit++) {
42        portENTER_CRITICAL(&s_lock);
43        gpio_set_level(DQ,0); esp_rom_delay_us(3);
44        gpio_set_level(DQ,1); esp_rom_delay_us(10);
45        if(gpio_get_level(DQ)) value|=1U<<bit;
46        esp_rom_delay_us(57);
47        portEXIT_CRITICAL(&s_lock);
48    }
49    return value;
50}
51uint8_t ds_crc(const uint8_t *bytes,int n)
52{
53    uint8_t crc=0;
54    while(n--) {
55        uint8_t value=*bytes++;
56        for(int i=0;i<8;i++) {
57            uint8_t mix=(crc^value)&1;
58            crc>>=1;
59            if(mix) crc^=0x8c; // Dallas/Maxim CRC-8，低位优先。
60            value>>=1;
61        }
62    }
63    return crc;
64}
65esp_err_t ds_read(float *celsius)
66{
67    if(!reset()) return ESP_ERR_NOT_FOUND;
68    write_byte(0xcc); write_byte(0x44); // 单个器件：Skip ROM，再开始转换。
69    vTaskDelay(pdMS_TO_TICKS(750)); // 12 位转换最大等待时间；此时不关闭中断。
70    if(!reset()) return ESP_ERR_NOT_FOUND;
71    write_byte(0xcc); write_byte(0xbe);
72    uint8_t b[9];
73    for(int i=0;i<9;i++) b[i]=read_byte();
74    if(ds_crc(b,8)!=b[8]) return ESP_ERR_INVALID_CRC;
75    // DS18B20 的保留字节应为 0xff、0x0c、0x10，排除全 0 假数据。
76    if(b[5]!=0xff || b[6]!=0x0c || b[7]!=0x10) return ESP_ERR_INVALID_RESPONSE;
77    float next=(int16_t)((b[1]<<8)|b[0])/16.0f;
78    if(next< -55 || next>125) return ESP_ERR_INVALID_RESPONSE;
79    *celsius=next;
80    return ESP_OK;
81}
